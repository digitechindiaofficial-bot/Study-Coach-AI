CREATE TABLE blog_posts (
  id UUID DEFAULT gen_random_uuid() 
     PRIMARY KEY,
  title TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  excerpt TEXT,
  content TEXT NOT NULL,
  cover_image TEXT,
  category TEXT NOT NULL 
    DEFAULT 'general',
  tags TEXT[] DEFAULT '{}',
  exam_code TEXT,
  author TEXT DEFAULT 'GovtGuru Team',
  is_published BOOLEAN DEFAULT false,
  is_featured BOOLEAN DEFAULT false,
  views INTEGER DEFAULT 0,
  read_time INTEGER DEFAULT 5,
  meta_title TEXT,
  meta_description TEXT,
  published_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE blog_posts 
  ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public read published" 
  ON blog_posts FOR SELECT 
  USING (is_published = true);

CREATE POLICY "Service role all access"
  ON blog_posts FOR ALL
  USING (auth.role() = 'service_role');

const AdminBlogList = () => {
  const [posts, setPosts] = useState([])
  const [filter, setFilter] = 
    useState('all')
  const [view, setView] = 
    useState('list') // 'list' or 'editor'
  const [editingPost, setEditingPost] = 
    useState(null)

  const fetchPosts = async () => {
    let query = supabase
      .from('blog_posts')
      .select('*')
      .order('created_at', 
             { ascending: false })

    if (filter === 'published') {
      query = query.eq('is_published', true)
    } else if (filter === 'draft') {
      query = query.eq('is_published', false)
    } else if (filter === 'featured') {
      query = query.eq('is_featured', true)
    }

    const { data } = await query
    setPosts(data || [])
  }

  const deletePost = async (id) => {
    if (!confirm('Delete this post?')) return
    await supabase
      .from('blog_posts')
      .delete()
      .eq('id', id)
    fetchPosts()
  }

  const togglePublish = async (post) => {
    await supabase
      .from('blog_posts')
      .update({ 
        is_published: !post.is_published,
        published_at: !post.is_published 
          ? new Date().toISOString() 
          : null
      })
      .eq('id', post.id)
    fetchPosts()
  }

  if (view === 'editor') {
    return <BlogEditor 
      post={editingPost}
      onBack={() => {
        setView('list')
        setEditingPost(null)
        fetchPosts()
      }}
    />
  }

  return (
    <div>
      <div style={{ display:'flex', 
                    justifyContent:'space-between',
                    marginBottom: 24 }}>
        <div>
          <h1>📝 Blog Manager</h1>
          <p>Manage blog posts for SEO</p>
        </div>
        <button onClick={() => {
          setEditingPost(null)
          setView('editor')
        }}>
          + Write New Post
        </button>
      </div>

      {/* Filter tabs */}
      <div style={{ display:'flex', gap: 8,
                    marginBottom: 16 }}>
        {['all','published','draft','featured']
          .map(f => (
          <button
            key={f}
            onClick={() => setFilter(f)}
            style={{
              background: filter === f 
                ? '#1B2A4A' : '#f1f5f9',
              color: filter === f 
                ? 'white' : '#64748b',
              padding: '6px 16px',
              borderRadius: 6,
              border: 'none',
              cursor: 'pointer',
              textTransform: 'capitalize'
            }}
          >
            {f}
          </button>
        ))}
      </div>

      {/* Posts table */}
      <table style={{ width:'100%', 
                      borderCollapse:'collapse' }}>
        <thead>
          <tr style={{ background:'#f8fafc' }}>
            <th>Title</th>
            <th>Category</th>
            <th>Status</th>
            <th>Views</th>
            <th>Date</th>
            <th>Actions</th>
          </tr>
        </thead>
        <tbody>
          {posts.map(post => (
            <tr key={post.id} 
                style={{ borderBottom: 
                         '1px solid #e2e8f0' }}>
              <td>
                <div style={{ fontWeight: 600 }}>
                  {post.title}
                </div>
                {post.is_featured && (
                  <span style={{ 
                    fontSize: 11,
                    background: '#FEF3C7',
                    color: '#92400E',
                    padding: '2px 6px',
                    borderRadius: 4
                  }}>
                    ⭐ Featured
                  </span>
                )}
              </td>
              <td>
                <span style={{
                  background: '#EFF6FF',
                  color: '#1D4ED8',
                  padding: '2px 8px',
                  borderRadius: 999,
                  fontSize: 12
                }}>
                  {post.category}
                </span>
              </td>
              <td>
                <span style={{
                  color: post.is_published 
                    ? '#10b981' : '#f59e0b'
                }}>
                  {post.is_published 
                    ? '✅ Published' 
                    : '📝 Draft'}
                </span>
              </td>
              <td>{post.views || 0}</td>
              <td>
                {post.published_at 
                  ? new Date(post.published_at)
                      .toLocaleDateString('en-IN')
                  : '—'}
              </td>
              <td>
                <div style={{ display:'flex',
                              gap: 8 }}>
                  <button onClick={() => {
                    setEditingPost(post)
                    setView('editor')
                  }}>✏️</button>
                  <button onClick={() => 
                    togglePublish(post)}>
                    {post.is_published 
                      ? '⬇️' : '🚀'}
                  </button>
                  <button 
                    style={{ color:'red' }}
                    onClick={() => 
                      deletePost(post.id)}>
                    🗑️
                  </button>
                </div>
              </td>
            </tr>
          ))}
        </tbody>
      </table>

      {posts.length === 0 && (
        <div style={{ textAlign:'center',
                      padding: '60px 20px',
                      color: '#94a3b8' }}>
          <div style={{ fontSize: 48 }}>📝</div>
          <p>No blog posts yet</p>
          <button onClick={() => 
            setView('editor')}>
            Write First Post
          </button>
        </div>
      )}
    </div>
  )
}

const BlogEditor = ({ post, onBack }) => {
  const [form, setForm] = useState({
    title: post?.title || '',
    slug: post?.slug || '',
    excerpt: post?.excerpt || '',
    content: post?.content || '',
    cover_image: post?.cover_image || '',
    category: post?.category || 'general',
    exam_code: post?.exam_code || '',
    author: post?.author || 'GovtGuru Team',
    tags: post?.tags || [],
    is_published: post?.is_published || false,
    is_featured: post?.is_featured || false,
    read_time: post?.read_time || 5,
    meta_title: post?.meta_title || '',
    meta_description: post?.meta_description || ''
  })

  const [tagInput, setTagInput] = useState('')
  const [saving, setSaving] = useState(false)
  const [toast, setToast] = useState(null)

  // Auto generate slug from title
  const generateSlug = (title) => {
    return title
      .toLowerCase()
      .replace(/[^a-z0-9\s-]/g, '')
      .replace(/\s+/g, '-')
      .replace(/-+/g, '-')
      .trim()
  }

  // Auto calculate read time
  const calculateReadTime = (content) => {
    const words = content
      .replace(/<[^>]*>/g, '')
      .split(/\s+/).length
    return Math.ceil(words / 200)
  }

  const handleTitleChange = (title) => {
    setForm(f => ({
      ...f,
      title,
      slug: !post ? generateSlug(title) : f.slug,
      meta_title: !f.meta_title 
        ? title + ' — GovtGuru' : f.meta_title
    }))
  }

  const addTag = () => {
    const tag = tagInput.trim().toLowerCase()
    if (tag && !form.tags.includes(tag)) {
      setForm(f => ({ 
        ...f, 
        tags: [...f.tags, tag] 
      }))
    }
    setTagInput('')
  }

  const removeTag = (tag) => {
    setForm(f => ({
      ...f,
      tags: f.tags.filter(t => t !== tag)
    }))
  }

  const savePost = async (publish = false) => {
    if (!form.title || !form.content) {
      setToast('Title and content required!')
      return
    }

    setSaving(true)

    const postData = {
      ...form,
      is_published: publish || form.is_published,
      published_at: publish 
        ? new Date().toISOString()
        : form.is_published 
          ? post?.published_at 
          : null,
      read_time: calculateReadTime(form.content),
      updated_at: new Date().toISOString()
    }

    if (post?.id) {
      // Update existing
      await supabase
        .from('blog_posts')
        .update(postData)
        .eq('id', post.id)
    } else {
      // Create new
      await supabase
        .from('blog_posts')
        .insert([postData])
    }

    setSaving(false)
    setToast(publish 
      ? '🚀 Post Published!' 
      : '✅ Draft Saved!')
    setTimeout(() => onBack(), 1500)
  }

  const CATEGORIES = [
    { value: 'bpsc', label: 'BPSC' },
    { value: 'uppsc', label: 'UPPSC' },
    { value: 'ssc', label: 'SSC' },
    { value: 'ibps', label: 'IBPS PO' },
    { value: 'sbi', label: 'SBI' },
    { value: 'rrb', label: 'RRB NTPC' },
    { value: 'current-affairs', 
      label: 'Current Affairs' },
    { value: 'study-tips', 
      label: 'Study Tips' },
    { value: 'general', label: 'General' },
  ]

  const EXAMS = [
    'BPSC', 'UPPSC', 'SSC_CGL', 
    'SSC_CHSL', 'IBPS_PO', 'IBPS_CLERK',
    'SBI_PO', 'SBI_CLERK', 'RRB_NTPC'
  ]

  return (
    <div style={{ maxWidth: 900, margin: '0 auto' }}>
      
      {/* Toast */}
      {toast && (
        <div style={{
          position: 'fixed',
          top: 20, right: 20,
          background: '#10b981',
          color: 'white',
          padding: '12px 20px',
          borderRadius: 8,
          fontWeight: 600,
          zIndex: 9999
        }}>
          {toast}
        </div>
      )}

      {/* Header */}
      <div style={{ display:'flex', 
                    justifyContent:'space-between',
                    alignItems:'center',
                    marginBottom: 24 }}>
        <div style={{ display:'flex', 
                      alignItems:'center',
                      gap: 12 }}>
          <button onClick={onBack}>
            ← Back
          </button>
          <h2>
            {post ? 'Edit Post' : 'New Blog Post'}
          </h2>
        </div>
        <div style={{ display:'flex', gap: 8 }}>
          <button 
            onClick={() => savePost(false)}
            disabled={saving}>
            {saving ? 'Saving...' : '💾 Save Draft'}
          </button>
          <button
            onClick={() => savePost(true)}
            disabled={saving}
            style={{ 
              background: '#1B2A4A',
              color: 'white'
            }}>
            🚀 Publish
          </button>
        </div>
      </div>

      {/* Two column layout */}
      <div style={{ display:'grid',
                    gridTemplateColumns:'1fr 300px',
                    gap: 24 }}>

        {/* LEFT — Main content */}
        <div style={{ display:'flex',
                      flexDirection:'column',
                      gap: 16 }}>

          {/* Title */}
          <div>
            <label>Blog Title *</label>
            <input
              value={form.title}
              onChange={e => 
                handleTitleChange(e.target.value)}
              placeholder="e.g. BPSC 2026 Complete Syllabus Guide"
              style={{ 
                width: '100%',
                fontSize: 20,
                fontWeight: 600,
                padding: '12px 16px',
                border: '1px solid #e2e8f0',
                borderRadius: 8
              }}
            />
          </div>

          {/* Slug */}
          <div>
            <label>URL Slug</label>
            <div style={{ display:'flex',
                          alignItems:'center',
                          gap: 8 }}>
              <span style={{ color:'#94a3b8',
                             fontSize: 14 }}>
                govtguru.com/blog/
              </span>
              <input
                value={form.slug}
                onChange={e => setForm(f => ({
                  ...f, slug: e.target.value
                }))}
                placeholder="bpsc-2026-syllabus-guide"
                style={{ flex: 1 }}
              />
            </div>
          </div>

          {/* Excerpt */}
          <div>
            <label>
              Excerpt (shown in blog list)
            </label>
            <textarea
              value={form.excerpt}
              onChange={e => setForm(f => ({
                ...f, excerpt: e.target.value
              }))}
              placeholder="Brief description of the post (150-160 chars for SEO)"
              rows={3}
              maxLength={160}

            />

                           color: '#94a3b8' }}>
              {form.excerpt.length}/160 chars
            </span>
          </div>

          {/* Content */}
          <div>
            <label>Content * (HTML supported)</label>
            <textarea

              onChange={e => setForm(f => ({
                ...f, content: e.target.value
              }))}
              placeholder={`Write your blog post here...

You can use HTML tags:

const BlogListPage = () => {
  const [posts, setPosts] = useState([])
  const [category, setCategory] = 
    useState('all')
  const [search, setSearch] = useState('')

  useEffect(() => {
    const fetchPosts = async () => {
      let query = supabase
        .from('blog_posts')
        .select(`id, title, slug, excerpt,
                 cover_image, category,
                 author, read_time, views,
                 published_at, is_featured,
                 exam_code, tags`)
        .eq('is_published', true)
        .order('published_at', 
               { ascending: false })

      if (category !== 'all') {
        query = query.eq('category', category)
      }

      if (search) {
        query = query.ilike('title', 
                            `%${search}%`)
      }

      const { data } = await query
      setPosts(data || [])
    }
    fetchPosts()
  }, [category, search])

  const featured = posts.find(
    p => p.is_featured
  )
  const regular = posts.filter(
    p => !p.is_featured
  )

  const categories = [
    { value: 'all', label: '📚 All' },
    { value: 'bpsc', label: '🎯 BPSC' },
    { value: 'uppsc', label: '🗺️ UPPSC' },
    { value: 'ssc', label: '📝 SSC' },
    { value: 'ibps', label: '🏦 IBPS' },
    { value: 'sbi', label: '💼 SBI' },

  const [post, setPost] = useState(null)
  const [related, setRelated] = useState([])

  useEffect(() => {
    const fetchPost = async () => {
      const { data } = await supabase
        .from('blog_posts')
        .select('*')
        .eq('slug', params.slug)
        .eq('is_published', true)
        .single()

      if (data) {
        setPost(data)
        
        // Increment views
        supabase
          .from('blog_posts')
          .update({ views: data.views + 1 })
          .eq('id', data.id)

        // Related posts
        const { data: rel } = await supabase
          .from('blog_posts')
          .select(`id, title, slug, 
                   excerpt, cover_image,
                   read_time, category`)
          .eq('is_published', true)
          .eq('category', data.category)
          .neq('id', data.id)
          .limit(3)
        
        setRelated(rel || [])
      }
