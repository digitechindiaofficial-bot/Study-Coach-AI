STEP 4 — Function to get topic name from tags:

function getTopicName(topicCode, tags) {
  if (!tags || tags.length === 0) {
    // Fallback: parse from topicCode
    // BPSC_GEO_001 → remove prefix → generic
    return topicCode || 'Unknown Topic'
  }

  // Tags to SKIP (too generic)
  const skipTags = [
    'easy', 'medium', 'hard',
    'history', 'geography', 'science', 
    'polity', 'economy', 'reasoning',
    'mathematics', 'english', 'hindi',
    'environment', 'general', 'india',
    'bihar', 'ancient', 'medieval', 'modern',
    'physics', 'chemistry', 'biology',
    'banking', 'current-affairs', 'mcq',
    'sahitya', 'vyakaran', 'alankar',
    'rivers', 'mountains', 'climate',
    'technology', 'computer'
  ]

  // Find the MOST SPECIFIC tag
  // Usually the 2nd or 3rd tag is most specific
  const specificTag = tags.find(
    t => !skipTags.includes(t.toLowerCase())
  )

  const tagToUse = specificTag || tags[1] || tags[0]
  
  if (!tagToUse) return 'General Topic'

  // Convert to readable Title Case
  // "kosi-river" → "Kosi River"
  // "kalinga-war" → "Kalinga War"
  // "flood-management" → "Flood Management"
  return tagToUse
    .replace(/_/g, '-')
    .split('-')
    .map(w => w.charAt(0).toUpperCase() + 
               w.slice(1).toLowerCase())
    .join(' ')
}

STEP 5 — Group by topicCode and calculate scores:

const topicStats = {}

attempts.forEach(attempt => {
  const code = attempt.topicCode
  if (!code) return

  if (!topicStats[code]) {
    const info = topicInfoMap[code] || {}
    topicStats[code] = {
      topicCode: code,
      topicName: getTopicName(code, info.tags),
      subjectCode: attempt.subjectCode || info.subjectCode,
      examCode: attempt.examCode,
      total: 0,
      correct: 0
    }
  }
  topicStats[code].total++
  if (attempt.is_correct) topicStats[code].correct++
})
