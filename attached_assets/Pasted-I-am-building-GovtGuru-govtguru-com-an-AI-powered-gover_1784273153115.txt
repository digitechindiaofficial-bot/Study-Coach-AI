CREATE TABLE IF NOT EXISTS exams (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  exam_code TEXT UNIQUE NOT NULL,
  exam_name TEXT NOT NULL,
  exam_full_name TEXT,
  category TEXT NOT NULL DEFAULT 'central',
  conducting_body TEXT,
  description TEXT,
  eligibility TEXT,
  exam_level TEXT DEFAULT 'national',
  target_state TEXT,
  is_active BOOLEAN DEFAULT true,
  is_featured BOOLEAN DEFAULT false,
  icon_emoji TEXT DEFAULT '📝',
  display_order INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS subjects (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  exam_code TEXT NOT NULL REFERENCES exams(exam_code) ON DELETE CASCADE,
  subject_code TEXT NOT NULL,
  subject_name TEXT NOT NULL,
  subject_full_name TEXT,
  syllabus_topics JSONB DEFAULT '[]',
  total_questions INTEGER DEFAULT 0,
  total_marks INTEGER,
  duration_minutes INTEGER,
  difficulty_level TEXT DEFAULT 'medium',
  is_active BOOLEAN DEFAULT true,
  display_order INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(exam_code, subject_code)
);

ALTER TABLE exams ENABLE ROW LEVEL SECURITY;
ALTER TABLE subjects ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read exams" ON exams FOR SELECT USING (true);
CREATE POLICY "Public read subjects" ON subjects FOR SELECT USING (true);
CREATE POLICY "Service role full access exams" ON exams FOR ALL USING (auth.role() = 'service_role');
CREATE POLICY "Service role full access subjects" ON subjects FOR ALL USING (auth.role() = 'service_role');
