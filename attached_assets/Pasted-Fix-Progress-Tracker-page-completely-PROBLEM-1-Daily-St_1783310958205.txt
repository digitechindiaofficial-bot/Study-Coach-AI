PROBLEM 2 - Day Streak not working:

Fix streak logic completely:

1. Streak increases when:
   - User completes at least 1 task today
   - AND studied yesterday too (consecutive days)

2. Streak resets to 0 when:
   - User misses a complete day (no tasks completed)

3. Streak logic in a function:

   const updateStreak = async (userId: string) => {
     const today = new Date().toDateString()
     const yesterday = new Date(
       Date.now() - 86400000
     ).toDateString()
     
     const { data: profile } = await supabase
       .from('profiles')
       .select('streak_count, last_active_date')
       .eq('id', userId)
       .single()
     
     const lastActive = new Date(
       profile.last_active_date
     ).toDateString()
     
     let newStreak = profile.streak_count
     
     if (lastActive === today) {
       // Already updated today, do nothing
       return
     } else if (lastActive === yesterday) {
       // Consecutive day - increment streak
       newStreak = profile.streak_count + 1
     } else {
       // Missed a day - reset streak
       newStreak = 1
     }
     
     await supabase
       .from('profiles')
       .update({ 
         streak_count: newStreak,
         last_active_date: new Date().toISOString()
       })
       .eq('id', userId)
   }

4. Call updateStreak() every time user completes a task.
