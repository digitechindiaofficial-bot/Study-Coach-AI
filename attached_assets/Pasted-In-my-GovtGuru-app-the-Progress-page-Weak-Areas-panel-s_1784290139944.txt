panel shows raw topic codes like "BPSC_SCI_008" 
instead of readable topic names.

Fix this by fetching topic names from the tags 
field in the question_bank table.

Find the Progress page component and in the section

STEP 1 — After fetching user's question_attempts,
get all unique topicCodes:

const uniqueTopicCodes = [
  ...new Set(
    attempts.map(a => a.topicCode).filter(Boolean)

STEP 2 — Fetch one question per topicCode from 
question_bank to get the tags:

const { data: topicQuestions } = await supabase
  .from('question_bank')
  .select('topicCode, tags')
  .in('topicCode', uniqueTopicCodes)

STEP 3 — Build a topicCode to name map using tags:

function getTopicNameFromTags(tags) {
  if (!tags || tags.length === 0) return null

  // Tags to skip — too generic
  const skipTags = [
    'easy', 'medium', 'hard', 'mcq',
    'banking', 'general', 'india', 'bihar',
    'history', 'geography', 'science', 'polity',
    'environment', 'economy', 'reasoning',
    'maths', 'mathematics', 'english', 'hindi',
    'ancient', 'medieval', 'modern',
    'physics', 'chemistry', 'biology',
    'technology', 'current-affairs',
    'sahitya', 'vyakaran', 'sahitya-itihas'
  ]

  // Find most specific tag (not in skip list)
  const bestTag = tags.find(
    t => !skipTags.includes(t.toLowerCase())
  )

  const tagToUse = bestTag || tags[0]
  if (!tagToUse) return null

  // Convert kebab-case or underscore to Title Case
  // "cell-biology" → "Cell Biology"
  // "number_series" → "Number Series"
  return tagToUse
    .replace(/_/g, '-')
    .split('-')
    .map(w => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase())
    .join(' ')
}

// Build the cache map
const topicNameMap = {}
topicQuestions?.forEach(q => {
  if (q.topicCode && !topicNameMap[q.topicCode]) {
    topicNameMap[q.topicCode] = getTopicNameFromTags(q.tags)
  }
})

STEP 4 — Group attempts by topicCode and 
calculate score per topic:

const topicStats = {}

attempts.forEach(attempt => {
  const code = attempt.topicCode
  if (!code) return
  
  if (!topicStats[code]) {
    topicStats[code] = {
      topicCode: code,
      topicName: topicNameMap[code] || code,
      subjectCode: attempt.subjectCode,
      examCode: attempt.examCode,
      total: 0,
      correct: 0
    }
  }
  topicStats[code].total++
  if (attempt.is_correct) topicStats[code].correct++
})

STEP 5 — Filter weak topics (score below 60%):

const weakTopics = Object.values(topicStats)
  .map(t => ({
    ...t,
    score: t.total > 0 
      ? Math.round((t.correct / t.total) * 100) 
      : 0
  }))
  .filter(t => t.total > 0 && t.score < 60)
  .sort((a, b) => a.score - b.score)
  .slice(0, 10)
