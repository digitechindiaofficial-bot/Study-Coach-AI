In the saveSettings function, after 
successfully saving new exam_code:

async function saveSettings() {
  // Get old exam before saving
  const oldExam = currentProfile?.exam_code
  const newExam = selectedExam  
  
  // Save to database
  await supabase
    .from('profiles')
    .update({ 
      exam_code: newExam,
      // other settings...
    })
    .eq('user_id', userId)
  
  // Check if exam changed
  if (oldExam !== newExam) {
    
    // DELETE old study plan from database
    await supabase
      .from('study_plans')
      .delete()
      .eq('user_id', userId)
    
    // Show message to user
    toast.success(
      'Exam changed! Generating new 
       study plan for ' + newExam + '...'
    )
    
    // Redirect to Study Planner
    // so new plan gets generated
    setTimeout(() => {
      window.location.href = '/planner'
    }, 1500)
    
  } else {
    // Exam not changed, normal save
    toast.success('Settings saved!')
  }
}

When the planner loads, check if 
saved plan matches current exam:

useEffect(() => {
  async function checkAndLoadPlan() {
    
    // Get user's current exam
    const { data: profile } = await supabase
      .from('profiles')
      .select('exam_code, exam_date, 
               daily_hours')
      .eq('user_id', userId)
      .single()
    
    const currentExam = profile?.exam_code
    
    // Get saved study plan
    const { data: savedPlan } = await supabase
      .from('study_plans')
      .select('*')
      .eq('user_id', userId)
      .single()
    
    // Check if plan exists AND 
    // matches current exam
    if (savedPlan && 
        savedPlan.exam_code === currentExam) {
      // Plan is valid — show it
      setPlan(savedPlan.plan_data)
      setCurrentExam(currentExam)
      
    } else {
      // Plan is outdated or missing
      // Auto generate new plan
      console.log('Exam changed or no plan found')
      console.log('Generating new plan for:', 
                  currentExam)
      
      // Delete old plan if exists
      if (savedPlan) {
        await supabase
          .from('study_plans')
          .delete()
          .eq('user_id', userId)
      }
      
      // Auto generate new plan
      await generateNewPlan(currentExam, profile)
    }
  }
  
  checkAndLoadPlan()
}, [userId])
