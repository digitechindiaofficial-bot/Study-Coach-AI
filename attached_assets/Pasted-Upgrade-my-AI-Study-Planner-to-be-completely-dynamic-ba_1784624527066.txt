STEP 2 — Calculate time per subject:

const totalSubjects = subjects.length
const dailyHours = profile.daily_hours || 4
const totalAvailableHours = 
  remainingDays * dailyHours

// Reserve last 20% days for revision
const revisionDays = Math.ceil(remainingDays * 0.20)
const studyDays = remainingDays - revisionDays

// Distribute study days among subjects
// Give more days to subjects with more topics
const totalTopics = subjects.reduce(
  (sum, s) => sum + (s.syllabus_topics?.length || 5), 
  0
)

const subjectPlan = subjects.map(sub => {
  const topicCount = sub.syllabus_topics?.length || 5
  const subjectDays = Math.max(
    Math.round((topicCount / totalTopics) * studyDays),
    3 // minimum 3 days per subject
  )
  const hoursPerDay = dailyHours
  const totalHours = subjectDays * hoursPerDay
  const daysPerTopic = subjectDays / topicCount

  return {
    subject: sub.subject_name,
    subjectCode: sub.subject_code,
    topics: sub.syllabus_topics || [],
    allocatedDays: subjectDays,
    totalHours: totalHours,
    daysPerTopic: Math.max(daysPerTopic, 0.5),
    weightage: Math.round(
      (topicCount / totalTopics) * 100
    )
  }
})

// Different plans based on remaining days

STEP 4 — Generate day-by-day calendar:

Create an actual calendar from today 
to exam date, assigning topics to each day:

const fullPlan = []
let currentDay = new Date(today)
let subjectIndex = 0
let topicIndex = 0

while (currentDay < examDate) {
  const dayOfWeek = currentDay.getDay()
  const isWeekend = dayOfWeek === 0 || dayOfWeek === 6
  const daysLeft = Math.ceil(
    (examDate - currentDay) / (1000*60*60*24)
  )
  const isLastWeek = daysLeft <= 7
  const isLastThreeDays = daysLeft <= 3

  let dayPlan = {
    date: currentDay.toISOString().split('T')[0],
    dayName: ['Sun','Mon','Tue','Wed',
               'Thu','Fri','Sat'][dayOfWeek],
    sessions: []
  }

  if (isLastThreeDays) {
    // Final 3 days — only revision
    dayPlan.sessions = [{
      type: "FINAL REVISION",
      topic: "Complete Syllabus Revision",
      subject: "All Subjects",
      duration: dailyHours * 60,
      tasks: [
        "Revise all your notes",
        "Solve previous year papers",
        "Focus on weak topics only",
        "Stay calm and confident"
      ]
    }]
  }
  else if (isLastWeek) {
    // Last week — mock tests
    dayPlan.sessions = [{
      type: "MOCK TEST",
      topic: "Full Mock Test",
      subject: "All Subjects", 
      duration: 120,
      tasks: [
        "Attempt full mock test",
        "Analyse mistakes",
        "Revise wrong answers"
      ]
    }]
  }
  else if (isWeekend) {
    // Weekend — revision of week's topics
    dayPlan.sessions = [{
      type: "REVISION",
      topic: "Weekly Revision",
      subject: "This week's subjects",
      duration: dailyHours * 60,
      tasks: [
        "Revise all topics studied this week",
        "Solve 50 mixed practice questions",
        "Make quick notes of important points"
      ]
    }]
  }
  else {
    // Weekday — study new topic
    const currentSubject = subjectPlan[subjectIndex]
    
    if (currentSubject && 
        topicIndex < currentSubject.topics.length) {
      
      const topic = currentSubject.topics[topicIndex]
      
      // Morning session — new topic
      dayPlan.sessions.push({
        type: "STUDY",
        time: "Morning",
        topic: topic,
        subject: currentSubject.subject,
        duration: Math.round(dailyHours * 60 * 0.6),
        tasks: [
          `Study: ${topic} in depth`,
          "Make concise notes",
          `Solve ${Math.round(dailyHours * 8)} MCQs`,
          "Note important facts and dates"
        ]
      })
      
      // Evening session — practice
      dayPlan.sessions.push({
        type: "PRACTICE",
        time: "Evening",
        topic: topic + " Practice",
        subject: currentSubject.subject,
        duration: Math.round(dailyHours * 60 * 0.4),
        tasks: [
          "Revise morning's notes",
          `Attempt ${Math.round(dailyHours * 5)} more MCQs`,
          "Review previous year questions on this topic",
          "Identify and note weak points"
        ]
      })

      // Move to next topic
      topicIndex++
      
      // If all topics of this subject done,
      // move to next subject
      if (topicIndex >= currentSubject.topics.length) {
        subjectIndex++
        topicIndex = 0
      }
    }
  }

  fullPlan.push(dayPlan)
  
  // Move to next day
  currentDay = new Date(currentDay)
  currentDay.setDate(currentDay.getDate() + 1)
}
