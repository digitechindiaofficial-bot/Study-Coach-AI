Step 1: Group question_attempts by topicCode:
const topicStats = {}
attempts.forEach(attempt => {
  const topic = attempt.topicCode
  if (!topicStats[topic]) {
    topicStats[topic] = { total: 0, correct: 0 }
  }
  topicStats[topic].total++
  if (attempt.is_correct) topicStats[topic].correct++

Step 2: Parse topic name from topicCode:
function parseTopicName(topicCode) {
  if (!topicCode) return 'Unknown Topic'
  // topicCode format: EXAM_SUBJ_TOPICNAME_NUM
  // e.g. "IBPS_PO_QA_NUMBER_SERIES_014"
  // Remove exam prefix and number suffix
  const parts = topicCode.split('_')
  // Remove first 3 parts (exam+subject) and last part (number)
  // IBPS_PO_QA_NUMBER_SERIES_014
  // parts: ['IBPS','PO','QA','NUMBER','SERIES','014']
  // Keep middle parts: NUMBER_SERIES → "Number Series"
  
  // Find where topic name starts (after subject code)
  // by checking known subject codes
  const subjectCodes = ['QA','ENG','GIR','GA','DESC',
    'BANK','COMP','RCA','MATH','HIST','POL','SCI',
    'HINDI','GEO','BIHAR','MREAS','NUM','REAS']
  
  let startIndex = -1
  for (let i = 0; i < parts.length; i++) {
    if (subjectCodes.includes(parts[i])) {
      startIndex = i + 1
      break
    }
  }
  
  if (startIndex === -1) return topicCode
  
  // Get topic parts, remove last numeric part
  const topicParts = parts.slice(startIndex)
  const lastPart = topicParts[topicParts.length - 1]
  const isNumeric = /^\d+$/.test(lastPart)
  if (isNumeric) topicParts.pop()
  
  // Convert to readable: NUMBER_SERIES → Number Series
  return topicParts
    .map(p => p.charAt(0).toUpperCase() + p.slice(1).toLowerCase())
    .join(' ')
}
