PROBLEM 3 - Streak not updating:

Fix streak logic:
1. When user completes ANY task, call this:

const updateStreak = async () => {
  const today = new Date().toISOString().split('T')[0]
  const yesterday = new Date(
    Date.now() - 86400000
  ).toISOString().split('T')[0]
  
  const { data: profile } = await supabase
    .from('profiles')
    .select('streak_count, last_active_date, best_streak')
    .eq('id', userId)
    .single()
  
  const lastActive = profile.last_active_date?.split('T')[0]
  
  if (lastActive === today) return // already updated
  
  let newStreak = 1
  if (lastActive === yesterday) {
    newStreak = (profile.streak_count || 0) + 1
  }
  
  const bestStreak = Math.max(
    newStreak, 
    profile.best_streak || 0
  )
  
  await supabase
    .from('profiles')
    .update({
      streak_count: newStreak,
      last_active_date: today,
      best_streak: bestStreak
    })
    .eq('id', userId)
}

2. Add best_streak column to profiles:

ADD COLUMN IF NOT EXISTS best_streak INTEGER DEFAULT 0;
