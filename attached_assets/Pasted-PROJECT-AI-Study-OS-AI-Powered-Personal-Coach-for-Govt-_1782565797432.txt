DATABASE SCHEMA (Supabase PostgreSQL)

Create these tables with proper RLS policies:

sql-- Users profile (extends Supabase auth.users)
CREATE TABLE profiles (
  id UUID REFERENCES auth.users PRIMARY KEY,
  full_name TEXT,
  exam_type TEXT CHECK (exam_type IN ('SSC_CGL','SSC_CHSL','IBPS_PO','IBPS_CLERK','SBI_PO','RRB_NTPC','UPPSC','BPSC','OTHER')),
  exam_date DATE,
  daily_study_hours INTEGER DEFAULT 4,
  plan_type TEXT DEFAULT 'free' CHECK (plan_type IN ('free','pro')),
  streak_count INTEGER DEFAULT 0,
  last_active_date DATE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- AI-generated study plans
CREATE TABLE study_plans (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  exam_type TEXT,
  plan_data JSONB, -- Full AI-generated plan
  weeks_remaining INTEGER,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Daily tasks
CREATE TABLE daily_tasks (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  date DATE DEFAULT CURRENT_DATE,
  subject TEXT,
  topic TEXT,
  duration_minutes INTEGER,
  task_type TEXT CHECK (task_type IN ('study','revision','quiz','current_affairs')),
  is_completed BOOLEAN DEFAULT FALSE,
  completed_at TIMESTAMPTZ,
  notes TEXT
);

-- Syllabus tracking per subject
CREATE TABLE syllabus_progress (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  exam_type TEXT,
  subject TEXT,
  topic TEXT,
  subtopic TEXT,
  status TEXT DEFAULT 'not_started' CHECK (status IN ('not_started','in_progress','completed')),
  confidence TEXT CHECK (confidence IN ('weak','medium','strong')),
  last_revised_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Current affairs articles
CREATE TABLE current_affairs (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  title TEXT NOT NULL,
  summary TEXT,
  category TEXT, -- Economy, Polity, Science, Sports, etc.
  exam_relevance TEXT[], -- ['SSC_CGL','IBPS_PO']
  published_date DATE DEFAULT CURRENT_DATE,
  source TEXT,
  is_featured BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Quiz questions
CREATE TABLE quiz_questions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  subject TEXT,
  topic TEXT,
  question_text TEXT NOT NULL,
  options JSONB, -- {a: "...", b: "...", c: "...", d: "..."}
  correct_option TEXT,
  explanation TEXT,
  difficulty TEXT CHECK (difficulty IN ('easy','medium','hard')),
  exam_type TEXT[],
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- User quiz attempts
CREATE TABLE quiz_attempts (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  question_id UUID REFERENCES quiz_questions(id),
  selected_option TEXT,
  is_correct BOOLEAN,
  time_taken_seconds INTEGER,
  attempted_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enable RLS on all tables
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE study_plans ENABLE ROW LEVEL SECURITY;
ALTER TABLE daily_tasks ENABLE ROW LEVEL SECURITY;
ALTER TABLE syllabus_progress ENABLE ROW LEVEL SECURITY;
ALTER TABLE quiz_attempts ENABLE ROW LEVEL SECURITY;

-- RLS: users can only see their own data
CREATE POLICY "Users own data" ON profiles FOR ALL USING (auth.uid() = id);
CREATE POLICY "Users own tasks" ON daily_tasks FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users own plan" ON study_plans FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users own progress" ON syllabus_progress FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Users own attempts" ON quiz_attempts FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "Public current affairs" ON current_affairs FOR SELECT USING (true);
CREATE POLICY "Public questions" ON quiz_questions FOR SELECT USING (true);

User clicks "Generate My Study Plan" button
App calls /api/generate-plan

Returns structured JSON study plan
